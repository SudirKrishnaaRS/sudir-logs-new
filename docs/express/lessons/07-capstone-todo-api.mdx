## Challenge: rebuild it from memory first

- `controllers/todoController.js` - reads req, validates `req.body`, calls the model, sends the right status code
- `routes/todos.js` - wires the 5 URLs to their controller functions via `express.Router()`
- `index.js` - logger middleware, `express.json()`, mounts routes, catch-all 404, error handler - in that order

## The complete, final code

`models/todoModel.js`:

```js title="models/todoModel.js"
let todos = [{ id: 1, title: 'Learn Express' }];
let nextId = 2;

module.exports = {
  getAll: () => todos,
  getById: (id) => todos.find(t => t.id === Number(id)),
  create: (data) => {
    const todo = { id: nextId++, title: data.title };
    todos.push(todo);
    return todo;
  },
  update: (id, data) => {
    const todo = todos.find(t => t.id === Number(id));
    if (!todo) return null;
    todo.title = data.title;
    return todo;
  },
  remove: (id) => {
    const index = todos.findIndex(t => t.id === Number(id));
    if (index === -1) return false;
    todos.splice(index, 1);
    return true;
  },
};
```

`controllers/todoController.js`:

```js title="controllers/todoController.js"
const todoModel = require('../models/todoModel');

function getAllTodos(req, res) {
  res.json(todoModel.getAll());
}

function getTodoById(req, res) {
  const todo = todoModel.getById(req.params.id);
  if (!todo) return res.status(404).json({ error: 'Todo not found' });
  res.json(todo);
}

function createTodo(req, res, next) {
  try {
    const { title } = req.body;
    if (!title) return res.status(400).json({ error: 'title is required' });
    res.status(201).json(todoModel.create({ title }));
  } catch (err) {
    next(err);
  }
}

function updateTodo(req, res, next) {
  try {
    const { title } = req.body;
    if (!title) return res.status(400).json({ error: 'title is required' });
    const updated = todoModel.update(req.params.id, { title });
    if (!updated) return res.status(404).json({ error: 'Todo not found' });
    res.json(updated);
  } catch (err) {
    next(err);
  }
}

function deleteTodo(req, res, next) {
  try {
    const deleted = todoModel.remove(req.params.id);
    if (!deleted) return res.status(404).json({ error: 'Todo not found' });
    res.status(204).end();
  } catch (err) {
    next(err);
  }
}

module.exports = { getAllTodos, getTodoById, createTodo, updateTodo, deleteTodo };
```

`routes/todos.js`:

```js title="routes/todos.js"
const express = require('express');
const router = express.Router();
const todoController = require('../controllers/todoController');

router.get('/', todoController.getAllTodos);
router.get('/:id', todoController.getTodoById);
router.post('/', todoController.createTodo);
router.put('/:id', todoController.updateTodo);
router.delete('/:id', todoController.deleteTodo);

module.exports = router;
```

`index.js`:

```js title="index.js"
const express = require('express');
const app = express();
const todoRoutes = require('./routes/todos');

app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next();
});
app.use(express.json());

app.use('/todos', todoRoutes);

app.use((req, res) => {
  res.status(404).json({ error: 'Route not found' });
});

app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: 'Something went wrong' });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
```

## Interview angle

<details>
  <summary>Walk me through what happens when a client sends `POST /todos` with `{"title": "Buy milk"}`.</summary>

  The logger middleware prints the method and URL, then calls `next()`. `express.json()` parses the body
  into `req.body`. Express matches the `/todos` mount, and the router sends `POST /` to `createTodo`, which
  validates `title`, calls `todoModel.create()`, and responds with 201 and the new todo. If anything throws,
  the catch block calls `next(err)` and the four-argument error handler responds with a JSON 500.
</details>

<details>
  <summary>Walk me through what happens when a client requests `GET /todos/999` versus `GET /nonexistent`.</summary>

  `GET /todos/999` matches the `/:id` route, so `getTodoById` runs, the model finds nothing, and the
  controller itself returns a 404 "Todo not found". `GET /nonexistent` matches no route at all, so the
  request falls through to the catch-all middleware after the routes, which returns a 404 "Route not found".
  Neither case reaches the error handler, because no error was passed to `next(err)`.
</details>

<details>
  <summary>Why split the app into routes, controllers and models if it all fits in one file?</summary>

  Each layer has one job: routes map URLs to functions, controllers handle the HTTP side (reading the
  request, validating, choosing status codes), and models own the data. That separation means swapping the
  in-memory array for a database only changes the model, while routes and controllers stay the same.
</details>
