## models/userModel.js

```js title="models/userModel.js"
const pool = require('../db/pool');

module.exports = {
  create: async ({ email, password }) => {
    const result = await pool.query(
      'INSERT INTO users (email, password) VALUES ($1, $2) RETURNING id, email',
      [email, password]
    );
    return result.rows[0];
  },
  findByEmail: async (email) => {
    const result = await pool.query('SELECT * FROM users WHERE email = $1', [email]);
    return result.rows[0];
  },
};
```

## controllers/authController.js

```js title="controllers/authController.js"
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');
const userModel = require('../models/userModel');

async function register(req, res, next) {
  try {
    const { email, password } = req.body;
    if (!email || !password) return res.status(400).json({ error: 'email and password are required' });

    const hashedPassword = await bcrypt.hash(password, 10);
    const user = await userModel.create({ email, password: hashedPassword });
    res.status(201).json(user);
  } catch (err) {
    next(err);
  }
}

async function login(req, res, next) {
  try {
    const { email, password } = req.body;
    const user = await userModel.findByEmail(email);
    if (!user) return res.status(401).json({ error: 'Invalid credentials' });

    const passwordMatches = await bcrypt.compare(password, user.password);
    if (!passwordMatches) return res.status(401).json({ error: 'Invalid credentials' });

    const token = jwt.sign({ userId: user.id }, process.env.JWT_SECRET, { expiresIn: '1h' });
    res.json({ token });
  } catch (err) {
    next(err);
  }
}

module.exports = { register, login };
```

## routes/auth.js and routes/todos.js

```js title="routes/auth.js"
// routes/auth.js
const express = require('express');
const router = express.Router();
const authController = require('../controllers/authController');

router.post('/register', authController.register);
router.post('/login', authController.login);
module.exports = router;
```

```js title="routes/todos.js"
// routes/todos.js
const express = require('express');
const { body } = require('express-validator');
const router = express.Router();
const requireAuth = require('../middleware/requireAuth');
const todoController = require('../controllers/todoController');

router.get('/', requireAuth, todoController.getAllTodos);
router.get('/:id', requireAuth, todoController.getTodoById);
router.post('/', requireAuth, body('title').notEmpty(), todoController.createTodo);
router.put('/:id', requireAuth, body('title').notEmpty(), todoController.updateTodo);
router.delete('/:id', requireAuth, todoController.deleteTodo);
module.exports = router;
```

## index.js

```js title="index.js"
require('dotenv').config();
const express = require('express');
const cors = require('cors');
const app = express();

app.use((req, res, next) => {
  console.log(`${req.method} ${req.url}`);
  next();
});
app.use(cors({ origin: 'http://localhost:5173' }));
app.use(express.json());

app.use('/auth', require('./routes/auth'));
app.use('/todos', require('./routes/todos'));

app.use((req, res) => res.status(404).json({ error: 'Route not found' }));

app.use((err, req, res, next) => {
  console.error(err.stack);
  res.status(500).json({ error: 'Something went wrong' });
});

app.listen(3000, () => console.log('Server running on http://localhost:3000'));
```

## Interview angle

<details>
  <summary>Walk me through what happens when a React app calls `GET /todos` with a token.</summary>

  The request passes through the middleware in `index.js` in order: the logger, `cors()` (which adds the
  `Access-Control-Allow-Origin` header the browser checks), then `express.json()`. It reaches the `/todos`
  router, where `requireAuth` verifies the JWT and sets `req.user` - or responds 401 and stops. The controller
  then calls the model, which runs a parameterized query through the `pg` pool, and the rows go back with `res.json()`.
  If anything throws, `next(err)` hands it to the four-argument error handler, which responds 500.
</details>

<details>
  <summary>Walk me through register and login. Where does each piece from Week 2 show up?</summary>

  `register` hashes the password with `bcrypt.hash(password, 10)` and stores only the hash; the `INSERT ... RETURNING id, email`
  query means the hash never goes back in the response. `login` looks the user up by email, checks the password with
  `bcrypt.compare`, and returns the same "Invalid credentials" 401 whether the email or the password was wrong. On success
  it signs a JWT containing `userId` with `JWT_SECRET` from `.env`, valid for one hour.
</details>

<details>
  <summary>Why is `index.js` ordered the way it is?</summary>

  Express runs middleware and routes in registration order. `cors()` and `express.json()` must come before the routers so
  every route gets the CORS header and a parsed `req.body`. The 404 handler goes after all routers so it only catches
  requests nothing matched, and the error handler goes last, identified by its four parameters `(err, req, res, next)`.
</details>
