```js
const { body, validationResult } = require('express-validator');

router.post('/', body('title').notEmpty(), (req, res, next) => {
  const errors = validationResult(req);
  if (!errors.isEmpty()) return res.status(400).json({ errors: errors.array() });
  next();
}, todoController.createTodo);
```

```js
const cors = require('cors');
app.use(cors({ origin: 'http://localhost:5173' })); // before your routes
```

<Callout type="warn" title="Always remember">
  CORS is enforced by the **browser**, not your server - `curl`/Postman never show it.
  `cors()` with no options allows every origin; pass an explicit one for production.
</Callout>
